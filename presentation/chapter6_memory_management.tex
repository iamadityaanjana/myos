\begin{frame}{Learning objectives}
  \begin{itemize}
    \item Understand why an OS needs heap allocation and virtual memory early
    \item Build a working kernel heap and \texttt{kmalloc}
    \item Enable paging safely with identity mapping
    \item Expose memory layout through a runtime memory map
    \item Connect memory features to real kernel subsystems
  \end{itemize}
  \concept{Target level}{
    Intermediate kernel memory management: practical, stable, and extensible.
  }
\end{frame}

\begin{frame}{Why memory management matters in kernels}
  \begin{itemize}
    \item Kernel structures grow quickly: IDT tables, queues, driver state
    \item Static arrays are inflexible and harder to evolve
    \item Paging provides address-translation foundation for future process isolation
    \item Memory map visibility is critical for debugging and correctness
  \end{itemize}
\end{frame}

\begin{frame}{What was implemented in this chapter}
  \begin{itemize}
    \item \texttt{memory.c}: heap init, \texttt{kmalloc}, \texttt{kmalloc\_aligned}, map export
    \item \texttt{paging.c}: page directory + first page table + \texttt{CR3}/\texttt{CR0.PG}
    \item Linker boundary symbols: \texttt{\_kernel\_start}, \texttt{\_kernel\_end}
    \item Core integrations: IDT allocation, keyboard queue allocation, paging table allocation
    \item Shell diagnostics: \texttt{heap}, \texttt{kmalloc}, \texttt{memmap}, \texttt{paging}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Heap + \texttt{kmalloc} design}
\begin{lstlisting}[style=clang]
#define KERNEL_HEAP_SIZE (128 * 1024)

void memory_init() {
    uint32_t kernel_end = (uint32_t)&_kernel_end;
    heap_start_addr = align_up(kernel_end, 16);
    heap_end_addr = heap_start_addr + KERNEL_HEAP_SIZE;
    heap_curr_addr = heap_start_addr;
}

void* kmalloc(uint32_t size) {
    size = align_up(size, 8);
    if (heap_curr_addr + size > heap_end_addr) return 0;
    void* result = (void*)heap_curr_addr;
    heap_curr_addr += size;
    return result;
}

void* kmalloc_aligned(uint32_t size, uint32_t align) {
    // used for paging structures and low-level tables
}
\end{lstlisting}
\concept{Teaching point}{
  A bump allocator is the simplest robust starting allocator for a freestanding kernel.
}
\end{frame}

\begin{frame}[fragile]{Paging implementation walkthrough}
\begin{lstlisting}[style=clang]
page_directory = kmalloc_aligned(4096, 4096);
first_page_table = kmalloc_aligned(4096, 4096);

for (int i = 0; i < 1024; i++) page_directory[i] = 0x00000002;
for (int i = 0; i < 1024; i++) first_page_table[i] = (i * 0x1000) | 0x3;

page_directory[0] = ((uint32_t)first_page_table) | 0x3;
mov page_directory -> CR3
set CR0.PG bit
\end{lstlisting}
\textbf{Result:}
\begin{itemize}
  \item Enables paging with identity mapping for first 4MB
  \item Uses allocator-backed structures, not compile-time static tables
  \item Keeps current kernel addresses valid while introducing translation layer
\end{itemize}
\end{frame}

\begin{frame}{Keyword glossary: paging terms}
  \small
  \begin{tabular}{@{}p{2.9cm}p{9.1cm}@{}}
    \toprule
    \textbf{Keyword} & \textbf{Meaning} \\
    \midrule
    \texttt{paging} & CPU memory translation from virtual to physical addresses. \\
    \texttt{page directory} & Top-level table of page table pointers. \\
    \texttt{page table} & Maps virtual pages to physical frames. \\
    \texttt{identity mapping} & Virtual address equals physical address. \\

    \texttt{CR0.PG} & Control bit enabling paging mechanism. \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{How memory is used by the OS core (not only shell)}
  \begin{itemize}
    \item \textbf{IDT storage}: allocated via \texttt{kmalloc\_aligned} in \texttt{idt\_init}
    \item \textbf{Keyboard queue}: allocated via \texttt{kmalloc} in \texttt{keyboard\_init}
    \item \textbf{Paging tables}: allocated via \texttt{kmalloc\_aligned} in \texttt{paging\_init}
    \item \textbf{Heap boundaries}: derived from linker symbols at boot
  \end{itemize}
  \concept{Conclusion}{
    Memory management is now part of kernel runtime infrastructure, not just command output.
  }
\end{frame}

\begin{frame}{Shell commands used for observation}
  \begin{itemize}
    \item Memory core: \texttt{heap}, \texttt{kmalloc N}, \texttt{memmap}, \texttt{paging}
    \item Time/date utilities: \texttt{time}, \texttt{set time HH:MM:SS}, \texttt{calendar}
    \item General utilities: \texttt{echo TEXT}, \texttt{reboot}, \texttt{shutdown}
  \end{itemize}
  \concept{Why shell mattered}{
    We turned memory internals into observable runtime behavior without external debugger tooling.
  }
\end{frame}

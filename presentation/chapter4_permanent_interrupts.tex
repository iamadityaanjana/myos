\begin{frame}{What problem we observed}
  \begin{itemize}
    \item Typing one key caused screen flicker/reset loops (5--6 times)
    \item System looked like it returned to BIOS/QEMU boot repeatedly
    \item Behavior matched a \textbf{triple fault} pattern, not a simple keyboard bug
  \end{itemize}
  \vspace{6pt}
  \warn{Temporary guards reduced symptoms, but did not remove the root cause.}
\end{frame}

\begin{frame}{Crash chain (from QEMU trace)}
  \begin{itemize}
    \item Hardware interrupt arrived on vector \texttt{0x21} (IRQ1 keyboard)
    \item CPU raised \texttt{\#GP} with error code \texttt{0x0008}
    \item Then \texttt{\#DF} occurred, then immediate \texttt{Triple fault}
  \end{itemize}
  \concept{Interpretation}{
    \texttt{err=0x0008} points to selector index 1 (kernel code segment, \texttt{CS=0x08}).
    This strongly indicates descriptor/GDT state failure while handling interrupts.
  }
\end{frame}

\begin{frame}{Keyword slide: exception terms}
  \small
  \begin{tabular}{@{}p{2.5cm}p{9.5cm}@{}}
    \toprule
    \textbf{Keyword} & \textbf{Meaning} \\
    \midrule
    \texttt{IRQ1} & Keyboard hardware interrupt line on the PIC. \\
    \texttt{Vector 0x21} & Interrupt number after PIC remap (IRQ1 \textrightarrow 33 decimal). \\
    \texttt{\#GP} & General Protection Fault: protection/descriptor violation. \\
    \texttt{\#DF} & Double Fault: second fault while delivering first fault. \\
    \texttt{Triple fault} & Fault during double-fault handling; CPU resets. \\
    \texttt{err=0x0008} & Fault references selector 0x08 (kernel code segment). \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}[fragile]{Fix 1: Kernel-owned GDT}
  \begin{lstlisting}[style=clang]
void gdt_init() {
    gdt_ptr.limit = sizeof(gdt) - 1;
    gdt_ptr.base  = (uint32_t)&gdt;

    gdt_set_gate(0, 0, 0,          0x00, 0x00);
    gdt_set_gate(1, 0, 0xFFFFFFFF, 0x9A, 0xCF); // code
    gdt_set_gate(2, 0, 0xFFFFFFFF, 0x92, 0xCF); // data

    gdt_flush((uint32_t)&gdt_ptr);
}
  \end{lstlisting}
  \textbf{Why this solved a key risk:}
  \begin{itemize}
    \item Stops relying on bootloader-era descriptor state at runtime
    \item Reloads \texttt{CS/DS/ES/FS/GS/SS} from a stable kernel GDT
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Keyword slide: GDT terms}
  \small
  \begin{tabular}{@{}p{2.5cm}p{9.5cm}@{}}
    \toprule
    \textbf{Keyword} & \textbf{Meaning} \\
    \midrule
    \texttt{GDT} & Global Descriptor Table: defines segment descriptors. \\
    \texttt{LGDT} & CPU instruction that loads the GDT pointer register. \\
    \texttt{Selector 0x08} & Entry 1 in GDT (kernel code segment). \\
    \texttt{Selector 0x10} & Entry 2 in GDT (kernel data segment). \\
    \texttt{0x9A} & Access byte for present, ring0, executable code segment. \\
    \texttt{0x92} & Access byte for present, ring0, writable data segment. \\
    \texttt{0xCF} & Granularity flags: 4KB granularity + 32-bit segment size. \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}[fragile]{Fix 2: Stable ISR/IRQ stack contract}
  \begin{lstlisting}[style=asm]
%macro ISR_NOERR 2
%1:
    push dword 0
    push dword %2
    jmp isr_common_stub
%endmacro

%macro ISR_ERR 2
%1:
    push dword %2
    jmp isr_common_stub
%endmacro

%macro IRQ_STUB 2
%1:
    push dword 0
    push dword %2
    jmp irq_common_stub
%endmacro
  \end{lstlisting}
  \textbf{Meaning:}
  \begin{itemize}
    \item Every handler builds the same metadata shape: \texttt{int\_no + err\_code}
    \item CPU error-code exceptions use \texttt{ISR\_ERR}
    \item IRQs always synthesize \texttt{err\_code=0}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Common Stub ABI (ASM \textrightarrow C)}
  \begin{lstlisting}[style=asm]
isr_common_stub:
    pusha
    cld
    push esp
    call isr_handler
    add esp, 4
    popa
    add esp, 8
    iretd

irq_common_stub:
    pusha
    cld
    push esp
    call irq_handler
    add esp, 4
    popa
    add esp, 8
    iretd
  \end{lstlisting}
  \concept{Why this matters}{
    A unified ABI prevents per-vector stack mismatches and keeps return via \texttt{iretd} deterministic.
  }
\end{frame}

\begin{frame}[fragile]{C Dispatcher Structures}
  \begin{lstlisting}[style=clang]
typedef struct {
    uint32_t edi, esi, ebp, esp;
    uint32_t ebx, edx, ecx, eax;
    uint32_t int_no, err_code;
    uint32_t eip, cs, eflags;
} registers_t;
  \end{lstlisting}
  \textbf{Logic behind order:}
  \begin{itemize}
    \item \texttt{pusha} register order first
    \item Stub metadata next: \texttt{int\_no}, \texttt{err\_code}
    \item CPU return frame last: \texttt{eip/cs/eflags}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Fix 3: Safer exception screen path}
  \begin{lstlisting}[style=clang]
void isr_handler(registers_t* regs) {
    volatile uint16_t* vga = (volatile uint16_t*)0xB8000;
    uint8_t color = 0x4F;
    ...
    vga[0] = (color << 8) | 'E';
    vga[1] = (color << 8) | 'X';
    vga[2] = (color << 8) | 'C';
    ...
    cli;
    while (1) { hlt; }
}
  \end{lstlisting}
  \warn{Critical point: on fatal exception, show minimal VGA marker and halt. No recursive high-level printing.}
\end{frame}

\begin{frame}[fragile]{Fix 4: Deferred keyboard processing}
  \begin{lstlisting}[style=clang]
void keyboard_irq_handler() {
    uint8_t sc = inb(0x60);
    queue_push(sc);
}

void keyboard_process_pending() {
    while (queue_not_empty()) {
        process_scancode(queue_pop());
    }
}

void irq_handler(registers_t* regs) {
    if (regs->int_no == 33) {
        keyboard_irq_handler();
    }

    if (regs->int_no >= 40) {
        outb(0xA0, 0x20);
    }
    outb(0x20, 0x20);
}
  \end{lstlisting}
  \textbf{Why this is safer:}
  \begin{itemize}
    \item IRQ context does minimal work (read port, queue data, EOI)
    \item Full scancode mapping and screen writes happen in main loop
    \item Reduces re-entrancy and stack pressure during interrupts
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Keyword slide: IRQ path terms}
  \small
  \begin{tabular}{@{}p{2.5cm}p{9.5cm}@{}}
    \toprule
    \textbf{Keyword} & \textbf{Meaning} \\
    \midrule
    \texttt{EOI} & End Of Interrupt command to PIC: allows next IRQ delivery. \\
    \texttt{PIC} & Programmable Interrupt Controller (8259A master/slave). \\
    \texttt{Remap} & Move IRQ vectors away from CPU exception range. \\
    \texttt{Top-half} & Fast IRQ part (minimal, non-blocking work). \\
    \texttt{Bottom-half} & Deferred processing outside IRQ context. \\
    \texttt{Scancode} & Raw keyboard code read from port \texttt{0x60}. \\
    \texttt{iretd} & Returns from interrupt restoring \texttt{EIP/CS/EFLAGS}. \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}[fragile]{IDT Wiring: Full 0--47 Coverage}
  \begin{lstlisting}[style=clang]
idt_set_gate(0,  (uint32_t)isr0);
...
idt_set_gate(31, (uint32_t)isr31);

idt_set_gate(32, (uint32_t)irq0);
...
idt_set_gate(47, (uint32_t)irq15);
  \end{lstlisting}
  \concept{Permanent improvement}{
    No empty vectors in the active interrupt space. Unexpected IRQ/exception now has deterministic behavior.
  }
\end{frame}

\begin{frame}[fragile]{PIC Remap Reliability}
  \begin{lstlisting}[style=clang]
outb(0x20, 0x11); io_wait();
outb(0xA0, 0x11); io_wait();
...
outb(0x21, 0xFD); io_wait();  // only IRQ1 unmasked
outb(0xA1, 0xFF); io_wait();  // all slave IRQs masked
  \end{lstlisting}
  \textbf{Why this matters:}
  \begin{itemize}
    \item \texttt{io\_wait()} avoids command timing races on PIC programming
    \item Mask policy keeps only keyboard IRQ enabled during early bring-up
  \end{itemize}
\end{frame}

\begin{frame}{Debugging method that found the root cause}
  \begin{itemize}
    \item Reproduced with scripted monitor input: \texttt{sendkey a}
    \item Logged interrupt path with QEMU \texttt{-d int,in\_asm}
    \item Mapped crash EIP to linked kernel symbol addresses
    \item Found \texttt{\#GP err=0x0008} on IRQ1 delivery path
  \end{itemize}
  \concept{Lesson}{
    In low-level kernels, deterministic reproduction + trace logs beat guess-based patching.
  }
\end{frame}
